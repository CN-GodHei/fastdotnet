namespace Fastdotnet.Plugin.AIAgent.Configuration

using Fastdotnet.Plugin.AIAgent.Dtos;

namespace Fastdotnet.Plugin.AIAgent.Services
{
    /// <summary>
    /// AI 代理服务接口
    /// </summary>
    public interface IAIAgentService
    {
        /// <summary>
        /// 创建 AI 代理
        /// </summary>
        Task<string> CreateAgentAsync(AgentCreateRequest request);
        
        /// <summary>
        /// 更新 AI 代理
        /// </summary>
        Task UpdateAgentAsync(string agentId, AgentUpdateRequest request);
        
        /// <summary>
        /// 删除 AI 代理
        /// </summary>
        Task DeleteAgentAsync(string agentId);
        
        /// <summary>
        /// 获取代理详情
        /// </summary>
        Task<AgentDetailDto?> GetAgentAsync(string agentId);
        
        /// <summary>
        /// 获取代理列表
        /// </summary>
        Task<PagedResult<AgentListItemDto>> GetAgentsAsync(AgentQueryRequest request);
        
        /// <summary>
        /// 执行代理任务（同步）
        /// </summary>
        Task<AgentResponse> ExecuteAsync(string agentId, string prompt, string? conversationId = null);
        
        /// <summary>
        /// 执行代理任务（流式）
        /// </summary>
        IAsyncEnumerable<string> ExecuteStreamingAsync(string agentId, string prompt, string? conversationId = null);
        
        /// <summary>
        /// 为代理添加插件
        /// </summary>
        Task AddPluginToAgentAsync(string agentId, string pluginId);
        
        /// <summary>
        /// 从代理移除插件
        /// </summary>
        Task RemovePluginFromAgentAsync(string agentId, string pluginId);
        
        /// <summary>
        /// 获取代理的插件列表
        /// </summary>
        Task<List<PluginInfoDto>> GetAgentPluginsAsync(string agentId);
        
        /// <summary>
        /// 启用/禁用代理
        /// </summary>
        Task ToggleAgentAsync(string agentId, bool enabled);
    }
}

using SqlSugar;

namespace Fastdotnet.Plugin.AIAgent.Entities
{
    /// <summary>
    /// AI 对话会话
    /// </summary>
    [SugarTable("AI_Conversations")]
    [SugarIndex("IX_AI_Conversations_UserId", nameof(UserId), OrderByType.Asc)]
    [SugarIndex("IX_AI_Conversations_AgentId", nameof(AgentId), OrderByType.Asc)]
    public class AIConversation
    {
        /// <summary>
        /// 主键 ID
        /// </summary>
        [SugarColumn(IsPrimaryKey = true)]
        public long Id { get; set; }
        
        /// <summary>
        /// 用户 ID
        /// </summary>
        [SugarColumn(IsNullable = false)]
        public long UserId { get; set; }
        
        /// <summary>
        /// 代理 ID
        /// </summary>
        [SugarColumn(Length = 100, IsNullable = false)]
        public string AgentId { get; set; } = "";
        
        /// <summary>
        /// 会话标题
        /// </summary>
        [SugarColumn(Length = 200, IsNullable = true)]
        public string? Title { get; set; }
        
        /// <summary>
        /// 状态：1-活跃 0-已归档
        /// </summary>
        [SugarColumn(IsNullable = false)]
        public int Status { get; set; } = 1;
        
        /// <summary>
        /// 创建时间
        /// </summary>
        [SugarColumn(IsNullable = false)]
        public DateTime CreatedTime { get; set; }
        
        /// <summary>
        /// 更新时间
        /// </summary>
        [SugarColumn(IsNullable = true)]
        public DateTime? UpdatedTime { get; set; }
        
        /// <summary>
        /// 最后消息时间
        /// </summary>
        [SugarColumn(IsNullable = true)]
        public DateTime? LastMessageTime { get; set; }
        
        /// <summary>
        /// 消息数量
        /// </summary>
        [SugarColumn(IsNullable = false)]
        public int MessageCount { get; set; } = 0;
        
        /// <summary>
        /// 总 Token 数
        /// </summary>
        [SugarColumn(IsNullable = false)]
        public int TotalTokens { get; set; } = 0;
    }
}

using Microsoft.AspNetCore.Mvc;
using Fastdotnet.Plugin.AIAgent.Services;
using Fastdotnet.Plugin.AIAgent.Dtos;

namespace Fastdotnet.Plugin.AIAgent.Controllers
{
    /// <summary>
    /// AI 代理管理 API
    /// </summary>
    [ApiController]
    [Route("api/[controller]")]
    public class AIAgentController : ControllerBase
    {
        private readonly IAIAgentService _agentService;
        
        public AIAgentController(IAIAgentService agentService)
        {
            _agentService = agentService;
        }
        
        /// <summary>
        /// 创建 AI 代理
        /// </summary>
        [HttpPost]
        public async Task<IActionResult> CreateAgent([FromBody] AgentCreateRequest request)
        {
            var agentId = await _agentService.CreateAgentAsync(request);
            return Ok(new { id = agentId });
        }
        
        /// <summary>
        /// 更新 AI 代理
        /// </summary>
        [HttpPut("{id}")]
        public async Task<IActionResult> UpdateAgent(string id, [FromBody] AgentUpdateRequest request)
        {
            await _agentService.UpdateAgentAsync(id, request);
            return NoContent();
        }
        
        /// <summary>
        /// 删除 AI 代理
        /// </summary>
        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteAgent(string id)
        {
            await _agentService.DeleteAgentAsync(id);
            return NoContent();
        }
        
        /// <summary>
        /// 获取代理详情
        /// </summary>
        [HttpGet("{id}")]
        public async Task<IActionResult> GetAgent(string id)
        {
            var agent = await _agentService.GetAgentAsync(id);
            if (agent == null)
                return NotFound();
            return Ok(agent);
        }
        
        /// <summary>
        /// 获取代理列表
        /// </summary>
        [HttpGet]
        public async Task<IActionResult> GetAgents([FromQuery] AgentQueryRequest request)
        {
            var result = await _agentService.GetAgentsAsync(request);
            return Ok(result);
        }
        
        /// <summary>
        /// 执行代理任务
        /// </summary>
        [HttpPost("{id}/execute")]
        public async Task<IActionResult> ExecuteAgent(string id, [FromBody] ExecuteRequest request)
        {
            var response = await _agentService.ExecuteAsync(id, request.Prompt, request.ConversationId);
            return Ok(response);
        }
        
        /// <summary>
        /// 为代理添加插件
        /// </summary>
        [HttpPost("{id}/plugins/{pluginId}")]
        public async Task<IActionResult> AddPlugin(string id, string pluginId)
        {
            await _agentService.AddPluginToAgentAsync(id, pluginId);
            return NoContent();
        }
        
        /// <summary>
        /// 从代理移除插件
        /// </summary>
        [HttpDelete("{id}/plugins/{pluginId}")]
        public async Task<IActionResult> RemovePlugin(string id, string pluginId)
        {
            await _agentService.RemovePluginFromAgentAsync(id, pluginId);
            return NoContent();
        }
        
        /// <summary>
        /// 获取代理的插件列表
        /// </summary>
        [HttpGet("{id}/plugins")]
        public async Task<IActionResult> GetAgentPlugins(string id)
        {
            var plugins = await _agentService.GetAgentPluginsAsync(id);
            return Ok(plugins);
        }
        
        /// <summary>
        /// 启用/禁用代理
        /// </summary>
        [HttpPatch("{id}/toggle")]
        public async Task<IActionResult> ToggleAgent(string id, [FromBody] ToggleRequest request)
        {
            await _agentService.ToggleAgentAsync(id, request.Enabled);
            return NoContent();
        }
    }
    
    public class ExecuteRequest
    {
        public string Prompt { get; set; } = "";
        public string? ConversationId { get; set; }
    }
    
    public class ToggleRequest

    }
