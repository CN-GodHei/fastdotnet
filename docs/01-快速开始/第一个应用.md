## 🔧 第一步：创建后端实体

### 1. 定义实体类

在 `backend/Fastdotnet.Core/Entities/` 目录下创建 `Task.cs`：

```csharp
using SqlSugar;
using System;

namespace Fastdotnet.Core.Entities
{
    /// <summary>
    /// 任务实体
    /// </summary>
    [SugarTable("tasks")]
    public class Task : BaseEntity
    {
        /// <summary>
        /// 任务标题
        /// </summary>
        [SugarColumn(Length = 200, IsNullable = false)]
        public string Title { get; set; }

        /// <summary>
        /// 任务描述
        /// </summary>
        [SugarColumn(Length = 1000, IsNullable = true)]
        public string? Description { get; set; }

        /// <summary>
        /// 任务状态：0-待办，1-进行中，2-已完成
        /// </summary>
        public int Status { get; set; } = 0;

        /// <summary>
        /// 优先级：1-低，2-中，3-高
        /// </summary>
        public int Priority { get; set; } = 2;

        /// <summary>
        /// 截止日期
        /// </summary>
        public DateTime? DueDate { get; set; }

        /// <summary>
        /// 创建人ID
        /// </summary>
        public string CreatedBy { get; set; }
    }
}
```

### 2. 注册实体

在数据库初始化时，实体会自动被发现并创建表结构。

## 📝 第二步：创建 DTO

在 `backend/Fastdotnet.Core/Dtos/` 目录下创建 `TaskDtos.cs`：

```csharp
using System;
using System.ComponentModel.DataAnnotations;

namespace Fastdotnet.Core.Dtos
{
    /// <summary>
    /// 创建任务请求
    /// </summary>
    public class CreateTaskRequest
    {
        [Required(ErrorMessage = "任务标题不能为空")]
        [StringLength(200, ErrorMessage = "标题长度不能超过200个字符")]
        public string Title { get; set; }

        [StringLength(1000, ErrorMessage = "描述长度不能超过1000个字符")]
        public string? Description { get; set; }

        public int Priority { get; set; } = 2;

        public DateTime? DueDate { get; set; }
    }

    /// <summary>
    /// 更新任务请求
    /// </summary>
    public class UpdateTaskRequest
    {
        public string? Title { get; set; }
        public string? Description { get; set; }
        public int? Status { get; set; }
        public int? Priority { get; set; }
        public DateTime? DueDate { get; set; }
    }

    /// <summary>
    /// 任务响应
    /// </summary>
    public class TaskResponse
    {
        public string Id { get; set; }
        public string Title { get; set; }
        public string? Description { get; set; }
        public int Status { get; set; }
        public int Priority { get; set; }
        public DateTime? DueDate { get; set; }
        public string CreatedBy { get; set; }
        public DateTime CreatedTime { get; set; }
        public DateTime? UpdatedTime { get; set; }
    }
}
```

## 💼 第三步：创建服务层

### 1. 定义服务接口

在 `backend/Fastdotnet.Service/IService/` 目录下创建 `ITaskService.cs`：

```csharp
using Fastdotnet.Core.Dtos;

namespace Fastdotnet.Service.IService
{
    public interface ITaskService
    {
        /// <summary>
        /// 创建任务
        /// </summary>
        Task<string> CreateAsync(CreateTaskRequest request);

        /// <summary>
        /// 获取任务列表
        /// </summary>
        Task<List<TaskResponse>> GetListAsync(int status = -1);

        /// <summary>
        /// 获取任务详情
        /// </summary>
        Task<TaskResponse?> GetByIdAsync(string id);

        /// <summary>
        /// 更新任务
        /// </summary>
        Task<bool> UpdateAsync(string id, UpdateTaskRequest request);

        /// <summary>
        /// 删除任务
        /// </summary>
        Task<bool> DeleteAsync(string id);
    }
}
```

### 2. 实现服务

在 `backend/Fastdotnet.Service/Service/` 目录下创建 `TaskService.cs`：

```csharp
using Fastdotnet.Core.Dtos;
using Fastdotnet.Core.Entities;
using Fastdotnet.Service.IService;
using SqlSugar;

namespace Fastdotnet.Service.Service
{
    public class TaskService : ITaskService
    {
        private readonly ISqlSugarClient _db;

        public TaskService(ISqlSugarClient db)
        {
            _db = db;
        }

        public async Task<string> CreateAsync(CreateTaskRequest request)
        {
            var task = new Core.Entities.Task
            {
                Id = Guid.NewGuid().ToString(),
                Title = request.Title,
                Description = request.Description,
                Priority = request.Priority,
                DueDate = request.DueDate,
                Status = 0,
                CreatedBy = "current-user-id", // TODO: 从当前用户获取
                CreatedTime = DateTime.Now
            };

            await _db.Insertable(task).ExecuteCommandAsync();
            return task.Id;
        }

        public async Task<List<TaskResponse>> GetListAsync(int status = -1)
        {
            var query = _db.Queryable<Core.Entities.Task>();

            if (status >= 0)
            {
                query = query.Where(t => t.Status == status);
            }

            var tasks = await query.OrderByDescending(t => t.CreatedTime).ToListAsync();

            return tasks.Select(t => new TaskResponse
            {
                Id = t.Id,
                Title = t.Title,
                Description = t.Description,
                Status = t.Status,
                Priority = t.Priority,
                DueDate = t.DueDate,
                CreatedBy = t.CreatedBy,
                CreatedTime = t.CreatedTime,
                UpdatedTime = t.UpdatedTime
            }).ToList();
        }

        public async Task<TaskResponse?> GetByIdAsync(string id)
        {
            var task = await _db.Queryable<Core.Entities.Task>()
                .Where(t => t.Id == id)
                .FirstAsync();

            if (task == null) return null;

            return new TaskResponse
            {
                Id = task.Id,
                Title = task.Title,
                Description = task.Description,
                Status = task.Status,
                Priority = task.Priority,
                DueDate = task.DueDate,
                CreatedBy = task.CreatedBy,
                CreatedTime = task.CreatedTime,
                UpdatedTime = task.UpdatedTime
            };
        }

        public async Task<bool> UpdateAsync(string id, UpdateTaskRequest request)
        {
            var task = await _db.Queryable<Core.Entities.Task>()
                .Where(t => t.Id == id)
                .FirstAsync();

            if (task == null) return false;

            if (request.Title != null) task.Title = request.Title;
            if (request.Description != null) task.Description = request.Description;
            if (request.Status.HasValue) task.Status = request.Status.Value;
            if (request.Priority.HasValue) task.Priority = request.Priority.Value;
            if (request.DueDate.HasValue) task.DueDate = request.DueDate;

            task.UpdatedTime = DateTime.Now;

            return await _db.Updateable(task).ExecuteCommandAsync() > 0;
        }

        public async Task<bool> DeleteAsync(string id)
        {
            return await _db.Deleteable<Core.Entities.Task>()
                .Where(t => t.Id == id)
                .ExecuteCommandAsync() > 0;
        }
    }
}
```

## 🌐 第四步：创建 API 控制器

在 `backend/Fastdotnet.WebApi/Controllers/` 目录下创建 `TaskController.cs`：

```csharp
using Fastdotnet.Core.Dtos;
using Fastdotnet.Service.IService;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace Fastdotnet.WebApi.Controllers
{
    /// <summary>
    /// 任务管理
    /// </summary>
    [ApiController]
    [Route("api/[controller]")]
    [Authorize]
    public class TaskController : ControllerBase
    {
        private readonly ITaskService _taskService;

        public TaskController(ITaskService taskService)
        {
            _taskService = taskService;
        }

        /// <summary>
        /// 创建任务
        /// </summary>
        [HttpPost]
        public async Task<IActionResult> Create([FromBody] CreateTaskRequest request)
        {
            var id = await _taskService.CreateAsync(request);
            return Ok(new { id });
        }

        /// <summary>
        /// 获取任务列表
        /// </summary>
        [HttpGet]
        public async Task<IActionResult> GetList([FromQuery] int status = -1)
        {
            var tasks = await _taskService.GetListAsync(status);
            return Ok(tasks);
        }

        /// <summary>
        /// 获取任务详情
        /// </summary>
        [HttpGet("{id}")]
        public async Task<IActionResult> GetById(string id)
        {
            var task = await _taskService.GetByIdAsync(id);
            if (task == null)
                return NotFound();

            return Ok(task);
        }

        /// <summary>
        /// 更新任务
        /// </summary>
        [HttpPut("{id}")]
        public async Task<IActionResult> Update(string id, [FromBody] UpdateTaskRequest request)
        {
            var result = await _taskService.UpdateAsync(id, request);
            if (!result)
                return NotFound();

            return NoContent();
        }

        /// <summary>
        /// 删除任务
        /// </summary>
        [HttpDelete("{id}")]
        public async Task<IActionResult> Delete(string id)
        {
            var result = await _taskService.DeleteAsync(id);
            if (!result)
                return NotFound();

            return NoContent();
        }
    }
}
```

## 🎨 第五步：创建前端页面

### 1. 创建 API 调用文件

在 `Web/fastdotnet-admin/src/api/` 目录下创建 `task.ts`：

```typescript
import request from '@/utils/request'

export interface TaskItem {
  id: string
  title: string
  description?: string
  status: number
  priority: number
  dueDate?: string
  createdTime: string
}

export interface CreateTaskParams {
  title: string
  description?: string
  priority?: number
  dueDate?: string
}

// 获取任务列表
export function getTaskList(status?: number) {
  return request.get<TaskItem[]>('/api/task', { params: { status } })
}

// 获取任务详情
export function getTaskById(id: string) {
  return request.get<TaskItem>(`/api/task/${id}`)
}

// 创建任务
export function createTask(data: CreateTaskParams) {
  return request.post<{ id: string }>('/api/task', data)
}

// 更新任务
export function updateTask(id: string, data: Partial<CreateTaskParams>) {
  return request.put(`/api/task/${id}`, data)
}

// 删除任务
export function deleteTask(id: string) {
  return request.delete(`/api/task/${id}`)
}
```

### 2. 创建任务列表页面

在 `Web/fastdotnet-admin/src/views/` 目录下创建 `task/index.vue`：

```vue
<template>
  <div class="task-container">
    <el-card>
      <template #header>
        <div class="card-header">
          <span>任务管理</span>
          <el-button type="primary" @click="showCreateDialog">
            新建任务
          </el-button>
        </div>
      </template>

      <!-- 筛选 -->
      <el-form :inline="true" class="filter-form">
        <el-form-item label="状态">
          <el-select v-model="filterStatus" placeholder="全部" clearable @change="loadTasks">
            <el-option label="待办" :value="0" />
            <el-option label="进行中" :value="1" />
            <el-option label="已完成" :value="2" />
          </el-select>
        </el-form-item>
      </el-form>

      <!-- 表格 -->
      <el-table :data="taskList" v-loading="loading">
        <el-table-column prop="title" label="标题" />
        <el-table-column prop="priority" label="优先级" width="100">
          <template #default="{ row }">
            <el-tag :type="getPriorityType(row.priority)">
              {{ getPriorityText(row.priority) }}
            </el-tag>
          </template>
        </el-table-column>
        <el-table-column prop="status" label="状态" width="100">
          <template #default="{ row }">
            <el-tag :type="getStatusType(row.status)">
              {{ getStatusText(row.status) }}
            </el-tag>
          </template>
        </el-table-column>
        <el-table-column prop="dueDate" label="截止日期" width="120" />
        <el-table-column label="操作" width="200">
          <template #default="{ row }">
            <el-button size="small" @click="editTask(row)">编辑</el-button>
            <el-button size="small" type="danger" @click="deleteTaskConfirm(row.id)">
              删除
            </el-button>
          </template>
        </el-table-column>
      </el-table>
    </el-card>

    <!-- 创建/编辑对话框 -->
    <el-dialog v-model="dialogVisible" :title="dialogTitle" width="500px">
      <el-form :model="form" :rules="rules" ref="formRef" label-width="80px">
        <el-form-item label="标题" prop="title">
          <el-input v-model="form.title" placeholder="请输入任务标题" />
        </el-form-item>
        <el-form-item label="描述">
          <el-input v-model="form.description" type="textarea" :rows="3" />
        </el-form-item>
        <el-form-item label="优先级">
          <el-select v-model="form.priority">
            <el-option label="低" :value="1" />
            <el-option label="中" :value="2" />
            <el-option label="高" :value="3" />
          </el-select>
        </el-form-item>
        <el-form-item label="截止日期">
          <el-date-picker v-model="form.dueDate" type="date" placeholder="选择日期" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" @click="submitForm">确定</el-button>
      </template>
    </el-dialog>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { getTaskList, createTask, updateTask, deleteTask } from '@/api/task'
import type { TaskItem, CreateTaskParams } from '@/api/task'

const loading = ref(false)
const taskList = ref<TaskItem[]>([])
const filterStatus = ref<number | undefined>(undefined)
const dialogVisible = ref(false)
const dialogTitle = ref('新建任务')
const formRef = ref()

const form = ref<CreateTaskParams>({
  title: '',
  description: '',
  priority: 2,
  dueDate: undefined
})

const rules = {
  title: [{ required: true, message: '请输入任务标题', trigger: 'blur' }]
}

// 加载任务列表
const loadTasks = async () => {
  loading.value = true
  try {
    taskList.value = await getTaskList(filterStatus.value)
  } catch (error) {
    ElMessage.error('加载任务列表失败')
  } finally {
    loading.value = false
  }
}

// 显示创建对话框
const showCreateDialog = () => {
  dialogTitle.value = '新建任务'
  form.value = { title: '', description: '', priority: 2, dueDate: undefined }
  dialogVisible.value = true
}

// 编辑任务
const editTask = (task: TaskItem) => {
  dialogTitle.value = '编辑任务'
  form.value = {
    title: task.title,
    description: task.description,
    priority: task.priority,
    dueDate: task.dueDate
  }
  dialogVisible.value = true
}

// 提交表单
const submitForm = async () => {
  await formRef.value.validate()
  
  try {
    if (dialogTitle.value === '新建任务') {
      await createTask(form.value)
      ElMessage.success('创建成功')
    } else {
      // TODO: 实现更新逻辑
      ElMessage.success('更新成功')
    }
    dialogVisible.value = false
    loadTasks()
  } catch (error) {
    ElMessage.error('操作失败')
  }
}

// 删除任务
const deleteTaskConfirm = async (id: string) => {
  await ElMessageBox.confirm('确定要删除这个任务吗？', '提示', {
    type: 'warning'
  })
  
  try {
    await deleteTask(id)
    ElMessage.success('删除成功')
    loadTasks()
  } catch (error) {
    ElMessage.error('删除失败')
  }
}

// 辅助函数
const getPriorityText = (priority: number) => {
  const map: Record<number, string> = { 1: '低', 2: '中', 3: '高' }
  return map[priority] || '中'
}

const getPriorityType = (priority: number) => {
  const map: Record<number, any> = { 1: 'info', 2: '', 3: 'danger' }
  return map[priority] || ''
}

const getStatusText = (status: number) => {
  const map: Record<number, string> = { 0: '待办', 1: '进行中', 2: '已完成' }
  return map[status] || '未知'
}

const getStatusType = (status: number) => {
  const map: Record<number, any> = { 0: 'info', 1: 'warning', 2: 'success' }
  return map[status] || 'info'
}

onMounted(() => {
  loadTasks()
})
</script>
```
